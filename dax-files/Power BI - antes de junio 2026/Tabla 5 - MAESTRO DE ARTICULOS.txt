## 16. Lista_Partes_Intercambiables_SAP = 
-- 1. Capturamos el NroParte de la fila actual donde estamos parados
VAR ParteFilaActual = '05. MaestroDeMateriales_2025'[Key_NroParte]

-- 2. Buscamos el ID del grupo para esa parte específica
VAR GrupoActual = 
    LOOKUPVALUE(
        '12.2. TablaMapeoIntercambiables - SAP'[ÍndiceGrupoAlternativos],
        '12.2. TablaMapeoIntercambiables - SAP'[NroParte],
        ParteFilaActual
    )

-- 3. Obtenemos todas las partes que pertenecen a ese ID de grupo
VAR TablaPartesDelGrupo = 
    CALCULATETABLE(
        VALUES('12.2. TablaMapeoIntercambiables - SAP'[NroParte]),
        '12.2. TablaMapeoIntercambiables - SAP'[ÍndiceGrupoAlternativos] = GrupoActual,
        ALL('03. InventarioActual - ReporteDeLogistica') -- Limpiamos filtros para ver todo el grupo
    )

-- 4. Quitamos la parte de la fila actual de la lista de resultados
VAR SoloAlternativas = 
    EXCEPT(
        TablaPartesDelGrupo, 
        { ParteFilaActual } 
    )

RETURN
IF(
    ISBLANK(GrupoActual), 
    BLANK(), 
    CONCATENATEX(
        SoloAlternativas, 
        '12.2. TablaMapeoIntercambiables - SAP'[NroParte], 
        UNICHAR(10), -- Puedes cambiarlo por UNICHAR(10) si quieres saltos de línea
        '12.2. TablaMapeoIntercambiables - SAP'[NroParte], ASC
    )
)

## 17. AeronavesDondeSeUtiliza_SegunMGA = 
-- 1. Capturamos la descripción de la fila actual (Inspección Completa)
VAR NroParteActual = '05. MaestroDeMateriales_2025'[Key_NroParte]

-- 2. Creamos una tabla virtual con todas las aeronaves asociadas a esa inspección específica
-- NOTA: Ajusta el nombre de la tabla de mapeo según la tengas en tu modelo
VAR TablaAeronaves = 
    CALCULATETABLE(
        VALUES('08.1. PatronDeInspecciones - MGA'[Aeronave]),
        '08.1. PatronDeInspecciones - MGA'[NroParte] = NroParteActual,
        ALL('08.1. PatronDeInspecciones - MGA') -- Limpiamos filtros para obtener la lista completa
    )

RETURN
-- 3. Concatenamos las matrículas de las aeronaves encontradas
IF(
    ISBLANK(NroParteActual), 
    "Sin descripción", 
    COALESCE(
        CONCATENATEX(
            TablaAeronaves, 
            '08.1. PatronDeInspecciones - MGA'[Aeronave], 
            ", ", -- Separador
            '08.1. PatronDeInspecciones - MGA'[Aeronave], ASC
        ),
        "No hay aeronaves mapeadas"
    )
)
